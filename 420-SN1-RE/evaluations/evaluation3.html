Méthodes numériques (ch. 12)

def bissection(f, a, b, tolerance):      # racine de f entre a et b
    while b - a > tolerance:             # f(a) et f(b) de signes contraires!
        m = (a + b) / 2
        if f(a) * f(m) <= 0:
            b = m
        else:
            a = m
    return (a + b) / 2
